rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isDeveloper() { return signedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.username == 'dev'; }
    function isMember(threadId) {
      return signedIn()
        && exists(/databases/$(database)/documents/threads/$(threadId))
        && request.auth.uid in get(/databases/$(database)/documents/threads/$(threadId)).data.participants;
    }
    function isGroupAdmin(threadId) {
      return signedIn() && (
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.username == 'Tuqa'
        || (get(/databases/$(database)/documents/threads/$(threadId)).data.admins is list
          && request.auth.uid in get(/databases/$(database)/documents/threads/$(threadId)).data.admins)
      );
    }

    match /stickers/{stickerId} {
      allow read: if signedIn() && !isDeveloper();
      allow create: if signedIn() && !isDeveloper()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.url is string
        && request.resource.data.url.matches('https://res\\.cloudinary\\.com/.*')
        && request.resource.data.label is string
        && request.resource.data.mediaType in ['image', 'gif'];
      allow update, delete: if false;
    }

    match /stories/{storyId} {
      allow read: if signedIn() && !isDeveloper();
      allow create: if signedIn() && !isDeveloper()
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.mediaType in ['image','video','text']
        && request.resource.data.caption is string
        && (request.resource.data.mediaType == 'text' || request.resource.data.mediaUrl is string);
      allow update, delete: if signedIn() && !isDeveloper() && resource.data.uid == request.auth.uid;
    }

    match /users/{userId} {
      allow read: if signedIn()
        && (request.auth.uid == userId
          || (resource.data.username in ['Samaa', 'Tuqa', 'Menna'] && !isDeveloper()));
      allow create, delete: if false;
      allow update: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'displayName','color','color1','color2','nameFont','decoration','photoURL',
          'bio','statusText','bannerURL','bannerHeight','avatarSize','isOnline',
          'presenceState','lastActiveAt','lastSeenAt'
        ]);
      match /fcmTokens/{tokenId} {
        allow read: if false;
        allow create, update: if signedIn()
          && request.auth.uid == userId
          && request.resource.data.keys().hasOnly(['token','platform','updatedAt'])
          && request.resource.data.token is string
          && request.resource.data.token.size() > 20
          && request.resource.data.platform in ['web','web-android','android-webview'];
        allow delete: if signedIn() && request.auth.uid == userId;
      }
    }

    match /system/maintenance {
      allow read: if signedIn();
      allow create, update: if signedIn()
        && isDeveloper()
        && request.resource.data.keys().hasOnly(['enabled','message','startedAt','endsAt','durationMinutes','openEnded','updatedAt']);
      allow delete: if false;
    }

    match /threads/{threadId} {
      // Allow signed-in users to check whether a deterministic thread ID exists.
      // Existing thread contents remain readable only by listed participants.
      // This lets a user create a new private thread after getDoc() returns missing.
      allow get: if signedIn() && !isDeveloper()
        && (!exists(/databases/$(database)/documents/threads/$(threadId))
          || request.auth.uid in resource.data.participants);
      allow list: if signedIn() && !isDeveloper() && request.auth.uid in resource.data.participants;
      allow create: if signedIn() && !isDeveloper()
        && request.resource.data.participants is list
        && request.auth.uid in request.resource.data.participants
        && (
          (request.resource.data.kind == 'self'
            && request.resource.data.participants.size() == 1
            && request.resource.data.participants[0] == request.auth.uid)
          || (request.resource.data.kind != 'self'
            && (request.resource.data.participants.size() == 2 || request.resource.data.participants.size() == 3))
        )
        && (request.resource.data.kind != 'group'
          || (request.resource.data.admins is list && request.resource.data.admins.size() == 0)
          || get(/databases/$(database)/documents/users/$(request.auth.uid)).data.username == 'Tuqa');
      allow update: if signedIn() && !isDeveloper()
        && request.auth.uid in resource.data.participants
        && request.resource.data.participants == resource.data.participants
        && (
          !request.resource.data.diff(resource.data).affectedKeys().hasAny(['groupName','groupPhotoURL','groupBannerURL','admins'])
          || (resource.data.kind == 'group' && isGroupAdmin(threadId))
        );
      allow delete: if false;
      match /typing/{userId} {
        allow read: if isMember(threadId);
        allow create, update: if isMember(threadId)
          && request.auth.uid == userId && request.resource.data.uid == request.auth.uid;
        allow delete: if isMember(threadId) && request.auth.uid == userId;
      }
      match /messages/{messageId} {
        allow read: if isMember(threadId);
        allow create: if isMember(threadId)
          && request.resource.data.senderId == request.auth.uid
          && (
            (request.resource.data.text is string
              && request.resource.data.text.size() > 0
              && request.resource.data.text.size() <= 2000)
            || request.resource.data.sticker is string
            || (request.resource.data.imageUrl is string
              && request.resource.data.imageUrl.matches('https://res\\.cloudinary\\.com/.*'))
          );
        allow update: if isMember(threadId)
          && request.resource.data.senderId == resource.data.senderId
          && request.resource.data.createdAt == resource.data.createdAt
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            'text','edited','editedAt','pinned','pinnedAt','reactions',
            'deletedFor','deletedForAll','deletedAt'
          ])
          && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['text'])
            || resource.data.senderId == request.auth.uid)
          && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['deletedForAll','deletedAt'])
            || (resource.data.senderId == request.auth.uid && request.resource.data.deletedForAll == true))
          && (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['deletedFor'])
            || (request.resource.data.deletedFor is list && request.auth.uid in request.resource.data.deletedFor));
        allow delete: if false;
      }
    }

  }
}
